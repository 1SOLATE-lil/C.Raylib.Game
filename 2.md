
#include "raylib.h"
#include <math.h>

int main(void) {
    InitWindow(800, 600, "Mini Danmaku");
    SetTargetFPS(60);

    Vector2 player = {400, 500};
    Vector2 enemy = {400, 100};
    Vector2 bullet[100];
    Vector2 shot[100];

    int score = 0;
    int gameOver = 0;

    for (int i = 0; i < 100; i++) {
        bullet[i] = (Vector2){-100, -100};
        shot[i] = (Vector2){-100, -100};
    }

    while (!WindowShouldClose()) {
        if (!gameOver) {
            if (IsKeyDown(KEY_LEFT)) player.x -= 4;
            if (IsKeyDown(KEY_RIGHT)) player.x += 4;
            if (IsKeyDown(KEY_UP)) player.y -= 4;
            if (IsKeyDown(KEY_DOWN)) player.y += 4;

            if (IsKeyPressed(KEY_SPACE)) {
                for (int i = 0; i < 100; i++) {
                    if (shot[i].y < 0) {
                        shot[i] = player;
                        break;
                    }
                }
            }

            enemy.x += sinf(GetTime()) * 2;

            for (int i = 0; i < 100; i++) {
                shot[i].y -= 6;

                if (shot[i].y < 0)
                    shot[i].y = -100;

                if (CheckCollisionCircles(shot[i], 5, enemy, 20)) {
                    shot[i].y = -100;
                    score++;
                }
            }

            if (GetRandomValue(0, 30) == 0) {
                for (int i = 0; i < 100; i++) {
                    if (bullet[i].y > 600) {
                        bullet[i] = enemy;
                        break;
                    }
                }
            }

            for (int i = 0; i < 100; i++) {
                bullet[i].y += 4;

                if (CheckCollisionCircles(player, 10, bullet[i], 6))
                    gameOver = 1;
            }
        }

        BeginDrawing();
        ClearBackground(BLACK);

        DrawCircleV(player, 10, BLUE);
        DrawCircleV(enemy, 20, RED);

        for (int i = 0; i < 100; i++) {
            DrawCircleV(bullet[i], 6, ORANGE);
            DrawCircleV(shot[i], 5, SKYBLUE);
        }

        DrawText(TextFormat("SCORE: %d", score), 10, 10, 20, WHITE);

        if (gameOver)
            DrawText("GAME OVER", 300, 280, 40, RED);

        EndDrawing();
    }

    CloseWindow();
    return 0;
}
